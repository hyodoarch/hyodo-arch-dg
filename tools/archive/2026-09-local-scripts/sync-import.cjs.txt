const fs=require('fs'),path=require('path');
const root=path.resolve(__dirname,'../..'),project=path.resolve(__dirname,'..');
const {report,downloads}=require('./project-import.json');
for(const r of report){
 const dst=path.join(project,'src/site/notes',r.target);
 fs.mkdirSync(path.dirname(dst),{recursive:true});
 if(!fs.existsSync(dst)) fs.copyFileSync(path.join(root,r.target),dst);
}
for(const [local] of downloads){
 const dst=path.join(project,'src/site/img/user',local);
 fs.mkdirSync(path.dirname(dst),{recursive:true});
 fs.copyFileSync(path.join(root,local),dst);
}
