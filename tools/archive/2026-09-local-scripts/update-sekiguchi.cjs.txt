const fs=require('fs');

const headings={
'sugito_IGP3021.jpg':'事務所・土間から通りを見る',
'sugito_IGP3056.jpg':'土間・事務所から住居エリアを見る',
'sugito_IGP2968.jpg':'リビング',
'sugito_IGP2982.jpg':'リビングから、土間、事務所を見る',
'sugito_IGP3055.jpg':'中庭',

'sugito-before_2013-09-18-12.40.jpg':'改装前の母屋外観',
'sugito-before_1F_2013-09-18-12.39.jpg':'改装前の事務所',

};
for(const file of paths){
 let src=fs.readFileSync(file,'utf8').replace(/\r\n/g,'\n');
 let count=0;
 src=src.replace(/```image-grid-captions\n[\s\S]*?```/g,block=>block.replace(/!\[\[([^|]+)\|([\s\S]*?)\]\]/g,(_,image,caption)=>{
   const name=image.split('/').pop(),heading=headings[name];
   if(!heading)throw Error(name);
   let body=caption.replace(/^###? [^\n]*\n/,'');
   if(body===caption && caption.startsWith(heading+'。'))body=caption.slice(heading.length+1);
   count++;
   return `![[${image}|## ${heading}\n${body}]]`;
 }));
 if(count!==16)throw Error('Expected 16 grids: '+count);
 // Only ordinary, standalone images outside grid blocks have these five filenames.
 const files='sugito_IGP(?:2925|2971|3046|3010|3086)\\.jpg';
 src=src.replace(new RegExp('!\\[\\[(images/sekiguchi/'+files+')(?:\\|[^\\]\\n]*)?\\]\\]','g'),'![[$1]]');
 src=src.replace(new RegExp('!\\[[^\\]\\n]*\\]\\(/img/user/(images/sekiguchi/'+files+')\\)','g'),'![[$1]]');
 fs.writeFileSync(file,src);
 console.log(file, count+' H2 captions');
}
