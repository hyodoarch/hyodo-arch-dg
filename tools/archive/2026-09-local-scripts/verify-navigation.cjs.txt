const {chromium}=require('../../.obsidian-image-grid-captions-dev/node_modules/@playwright/test');
const assert=require('node:assert/strict');
(async()=>{
 const browser=await chromium.launch({channel:'msedge',headless:true});
 try {
  const page=await browser.newPage();
  for(const route of ['/', '/projects/sekiguchi/']) {
   await page.goto('http://localhost:8080'+route,{waitUntil:'domcontentloaded'});
   for(const width of [1366,1920,1440,1000,390]) {
    await page.setViewportSize({width,height:1000});
    await page.waitForTimeout(500);
    // External Alpine may be unavailable in this verification browser.
    // Reproduce its desktop visibility without modifying project files.
    await page.locator('.filetree-wrapper').evaluate((el,w)=>el.style.setProperty('display',w>=1000?'flex':'none','important'),width);
    const r=await page.evaluate(()=>{
     const rect=s=>{const r=document.querySelector(s).getBoundingClientRect();return {left:r.left,right:r.right,width:r.width};};
     return {width:innerWidth,scroll:document.documentElement.scrollWidth,tree:rect('.filetree-wrapper'),main:rect('main.content')};
    });
    assert.ok(r.scroll<=width+1,JSON.stringify(r));
    if(width>=1000){assert.ok(r.tree.left>=r.main.right,JSON.stringify(r));assert.ok(Math.abs(r.tree.right-width)<20,JSON.stringify(r));}
    console.log(route,JSON.stringify(r));
    if(route==='/'&&width===1366)await page.screenshot({path:'.cache/navigation-right.png'});
   }
  }
 }finally{await browser.close();}
})().catch(e=>{console.error(e);process.exitCode=1;});
